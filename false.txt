
local Players = game:GetService("Players")
local HttpService = game:GetService("HttpService")

local LOCAL_PLAYER = Players.LocalPlayer

local VERIFY_URL = "https://smoothhubgetkey.smoothhubthailand.workers.dev/verify"
local SCRIPT_URL = "https://smoothhubgetkey.smoothhubthailand.workers.dev/script"

local KEY_FILE = "SmoothHub_key.txt"
local VERIFIED_FILE = "SmoothHub_verified.txt"
local SCRIPT_CACHE_FILE = "SmoothHub_AOTR_cache_v2.lua"
local SCRIPT_CACHE_TIME_FILE = "SmoothHub_AOTR_cache_v2.lua.t"

local VERIFIED_CACHE_SECONDS = 23 * 60 * 60
local SCRIPT_REFRESH_SECONDS = 10 * 60
local MIN_SCRIPT_SIZE = 100000

local KeySystem = {
    currentKey = nil,
    currentScript = nil,
}

-- Wait for Roblox to finish loading and for the local player to exist.
local function waitForGame(timeout, waitForCharacter)
    timeout = timeout or 30
    local startedAt = os.clock()

    while not game:IsLoaded() and os.clock() - startedAt < timeout do
        task.wait(0.1)
    end

    while not Players.LocalPlayer and os.clock() - startedAt < timeout do
        task.wait(0.1)
    end

    local player = Players.LocalPlayer
    if not player then
        return false
    end

    while not player:FindFirstChildOfClass("PlayerGui")
        and os.clock() - startedAt < timeout do
        task.wait(0.1)
    end

    if waitForCharacter then
        local characterStartedAt = os.clock()
        while not player.Character and os.clock() - characterStartedAt < 4 do
            task.wait(0.1)
        end
    end

    return true
end

-- Get the executor/Roblox identifier used by the original key server.
local function getHardwareId()
    local identifier

    pcall(function()
        identifier = game:GetService("RbxAnalyticsService"):GetClientId()
    end)

    if type(identifier) ~= "string" or #identifier < 8 then
        pcall(function()
            local getHwid = gethwid or get_hwid
            getHwid = getHwid or (syn and syn.gethwid)

            if getHwid then
                identifier = getHwid()
            end
        end)
    end

    identifier = tostring(identifier or ""):gsub("[^%w_-]", "")
    if #identifier < 8 then
        identifier = "unknown000"
    end

    return identifier:sub(1, 80)
end

-- HTTP GET with support for common executor request APIs.
local function httpGet(url)
    local ok, body = pcall(function()
        return game:HttpGet(url)
    end)

    if ok and type(body) == "string" and body ~= "" then
        return true, body
    end

    local requestFunction = syn and syn.request
        or http_request
        or request
        or (http and http.request)
        or (fluxus and fluxus.request)

    if requestFunction then
        local requestOk, response = pcall(requestFunction, {
            Url = url,
            Method = "GET",
        })

        if requestOk and type(response) == "table" then
            local status = response.StatusCode or response.Status or 200

            if type(response.Body) == "string"
                and response.Body ~= ""
                and (status == 200 or status == nil) then
                return true, response.Body
            end

            return false, "HTTP " .. tostring(status)
        end
    end

    return false, tostring(body)
end

-- Store the key in the executor environment, as the original did.
local function exposeKeyToEnvironment(key)
    pcall(function()
        if getgenv then
            getgenv().script_key = key
        end
    end)

    pcall(function()
        _G.script_key = key
    end)

    pcall(function()
        shared.script_key = key
    end)
end

local function buildRequestUrl(baseUrl, key)
    local userId = tostring(LOCAL_PLAYER.UserId)
    local hwid = getHardwareId()

    return baseUrl
        .. "?key=" .. HttpService:UrlEncode(key)
        .. "&uid=" .. HttpService:UrlEncode(userId)
        .. "&hwid=" .. HttpService:UrlEncode(hwid)
end

-- Verify one key against the remote service.
function KeySystem.verify(key)
    key = tostring(key or ""):gsub("%s", "")
    if key == "" then
        return false, "empty key"
    end

    if not LOCAL_PLAYER then
        return false, "local player unavailable"
    end

    KeySystem.currentKey = key

    local url = buildRequestUrl(VERIFY_URL, key)
    local lastReason = "denied"

    for attempt = 1, 3 do
        local requestOk, responseBody = httpGet(url)
        if not requestOk then
            return false, "server error"
        end

        local decodeOk, response = pcall(
            HttpService.JSONDecode,
            HttpService,
            responseBody
        )

        if not decodeOk or type(response) ~= "table" then
            return false, "server error"
        end

        if response.ok == true then
            exposeKeyToEnvironment(key)
            return true
        end

        lastReason = tostring(response.reason or "denied")

        if not lastReason:find("server busy", 1, true) or attempt == 3 then
            return false, lastReason
        end

        task.wait(0.5 + math.random() * 0.7)
    end

    return false, lastReason
end

-- Retry verification when the server reports temporary congestion.
function KeySystem.verifyWithRetries(key, attempts)
    attempts = attempts or 2
    local lastReason = "server error"

    for attempt = 1, attempts do
        local valid, reason = KeySystem.verify(key)
        if valid then
            return true
        end

        lastReason = reason
        if reason ~= "server error"
            and not tostring(reason):find("server busy", 1, true) then
            return false, reason
        end

        if attempt < attempts then
            task.wait(1 + attempt)
        end
    end

    return false, lastReason
end

-- Check whether a downloaded response looks like an HTML error page.
local function describeWebError(body)
    local title = body:match("<[Tt][Ii][Tt][Ll][Ee][^>]*>%s*(.-)%s*</[Tt][Ii][Tt][Ll][Ee]>")
    local errorCode = body:match("[Ee]rror code:?%s*(%d+)")
        or body:match("[Ee]rror%s+(1%d%d%d)")

    local details = {}
    if errorCode then
        table.insert(details, "code " .. errorCode)
    end
    if title and title ~= "" then
        table.insert(details, title:sub(1, 40))
    end

    return #details > 0 and (" [" .. table.concat(details, " | ") .. "]") or ""
end

-- Download and compile the main script, but do not execute it yet.
local function downloadMainScript()
    local lastError = "unknown"

    for attempt = 1, 3 do
        local url = buildRequestUrl(SCRIPT_URL, KeySystem.currentKey)

        if attempt > 1 then
            url = url .. "&_=" .. tostring(os.time()) .. tostring(attempt)
        end

        local requestOk, responseBody = httpGet(url)
        if not requestOk then
            lastError = "download failed: " .. tostring(responseBody)

            if tostring(responseBody):find("404", 1, true) then
                return nil, lastError
            end

            task.wait(1.5 * attempt)
            continue
        end

        responseBody = tostring(responseBody)
        local preview = responseBody:sub(1, 400):lower()

        if preview:find("<!doctype", 1, true)
            or preview:find("<html", 1, true) then
            return nil, "script URL returned a web page"
                .. describeWebError(responseBody)
        end

        if preview:find("not found", 1, true) and #responseBody < 300 then
            return nil, "script URL returned 'Not found'"
        end

        if #responseBody < MIN_SCRIPT_SIZE then
            lastError = "downloaded script is too short (" .. #responseBody .. " bytes)"
            task.wait(2 * attempt)
            continue
        end

        local compiler = loadstring or load
        if not compiler then
            return nil, "executor does not provide loadstring/load"
        end

        local compiledFunction = compiler(responseBody)
        if type(compiledFunction) == "function" then
            return responseBody
        end

        lastError = "compile error"
        task.wait(2 * attempt)
    end

    return nil, lastError
end

local function readCachedScript()
    local cachedScript

    pcall(function()
        if isfile and isfile(SCRIPT_CACHE_FILE) then
            cachedScript = readfile(SCRIPT_CACHE_FILE)
        end
    end)

    if type(cachedScript) == "string" and #cachedScript >= MIN_SCRIPT_SIZE then
        return cachedScript
    end

    pcall(function()
        if delfile and isfile and isfile(SCRIPT_CACHE_FILE) then
            delfile(SCRIPT_CACHE_FILE)
        end
    end)

    return nil
end

local function writeCachedScript(scriptSource)
    if type(scriptSource) ~= "string" or #scriptSource < MIN_SCRIPT_SIZE then
        return
    end

    pcall(function()
        if writefile then
            writefile(SCRIPT_CACHE_FILE, scriptSource)
        end
    end)
end

local function getCacheTimestamp()
    local timestamp = 0

    pcall(function()
        if isfile and isfile(SCRIPT_CACHE_TIME_FILE) then
            timestamp = tonumber(readfile(SCRIPT_CACHE_TIME_FILE)) or 0
        end
    end)

    return timestamp
end

-- Fetch a fresh script; use a previously cached copy only if fetching fails.
function KeySystem.getMainScript()
    if not KeySystem.currentKey then
        return nil, "verify a key first"
    end

    local cachedScript = readCachedScript()
    local compiler = loadstring or load

    if cachedScript and compiler then
        local cachedFunction = compiler(cachedScript)

        if type(cachedFunction) == "function" then
            if os.time() - getCacheTimestamp() >= SCRIPT_REFRESH_SECONDS then
                pcall(function()
                    if writefile then
                        writefile(SCRIPT_CACHE_TIME_FILE, tostring(os.time()))
                    end
                end)

                task.spawn(function()
                    local freshScript = downloadMainScript()
                    if freshScript and freshScript ~= cachedScript then
                        writeCachedScript(freshScript)
                    end
                end)
            end

            return cachedFunction
        end
    end

    local downloadedScript, downloadError = downloadMainScript()
    if downloadedScript then
        writeCachedScript(downloadedScript)
        return compiler(downloadedScript)
    end

    if cachedScript and compiler then
        local cachedFunction = compiler(cachedScript)
        if type(cachedFunction) == "function" then
            warn("[Smooth Hub] Server unreachable; using cached script.")
            return cachedFunction
        end
    end

    return nil, downloadError
end

-- Execute the downloaded/cached main script.
-- Review the remote script before calling this function.
function KeySystem.loadMainScript()
    local scriptFunction, loadError = KeySystem.getMainScript()
    if not scriptFunction then
        return false, loadError
    end

    waitForGame(30, true)

    local ok, result = pcall(scriptFunction)
    if not ok and tostring(result):find("attempt to index nil", 1, true) then
        waitForGame(30, true)
        task.wait(1.5)
        ok, result = pcall(scriptFunction)
    end

    if not ok then
        return false, tostring(result)
    end

    return true
end

-- Save/load the key. These files contain the key in plaintext.
function KeySystem.saveKey(key)
    key = tostring(key or ""):gsub("%s", "")
    if key == "" then
        return false
    end

    KeySystem.currentKey = key
    exposeKeyToEnvironment(key)

    local ok = pcall(function()
        if writefile then
            writefile(KEY_FILE, key)
            writefile(VERIFIED_FILE, tostring(os.time()) .. "|" .. key)
        end
    end)

    return ok
end

function KeySystem.readSavedKey()
    local savedKey

    pcall(function()
        if isfile and isfile(KEY_FILE) then
            savedKey = tostring(readfile(KEY_FILE)):gsub("%s", "")
        end
    end)

    return savedKey
end

function KeySystem.savedVerificationAge(key)
    local age

    pcall(function()
        if not isfile or not isfile(VERIFIED_FILE) then
            return
        end

        local timestamp, storedKey =
            tostring(readfile(VERIFIED_FILE)):match("^(%d+)|(.+)$")

        timestamp = tonumber(timestamp)
        if timestamp and storedKey == key then
            age = os.time() - timestamp
        end
    end)

    return age
end

function KeySystem.isSavedVerificationFresh(key)
    local age = KeySystem.savedVerificationAge(key)
    return age ~= nil and age >= 0 and age < VERIFIED_CACHE_SECONDS
end

function KeySystem.clearSavedVerification()
    pcall(function()
        if writefile then
            writefile(VERIFIED_FILE, "")
        end
    end)
end

-- Typical integration:
--[[
local key = KeySystem.readSavedKey()

if key then
    local valid = KeySystem.verifyWithRetries(key, 2)
    if valid then
        local loaded, err = KeySystem.loadMainScript()
        if not loaded then
            warn("[Smooth Hub] " .. tostring(err))
        end
    end
end

-- For a Submit button:
local valid, reason = KeySystem.verifyWithRetries(keyFromTextBox, 2)
if valid then
    KeySystem.saveKey(keyFromTextBox)
    local loaded, err = KeySystem.loadMainScript()
    if not loaded then
        warn("[Smooth Hub] " .. tostring(err))
    end
else
    warn("Key rejected: " .. tostring(reason))
end
]]

return KeySystem
